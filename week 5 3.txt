import java.util.*;

public class Solution {

    public static List<Integer> maxSubarray(List<Integer> arr) {

        // Maximum subarray
        int currentSum = arr.get(0);
        int maxSubarray = arr.get(0);

        // Maximum subsequence
        int maxSubsequence = 0;
        int maxElement = arr.get(0);

        for (int i = 0; i < arr.size(); i++) {

            int value = arr.get(i);

            // Kadane's algorithm
            if (i > 0) {
                currentSum = Math.max(value, currentSum + value);
                maxSubarray = Math.max(maxSubarray, currentSum);
            }

            // Track largest element
            maxElement = Math.max(maxElement, value);

            // Add all positive values
            if (value > 0) {
                maxSubsequence += value;
            }
        }

        // If all numbers are negative
        if (maxSubsequence == 0) {
            maxSubsequence = maxElement;
        }

        return Arrays.asList(maxSubarray, maxSubsequence);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();

            List<Integer> arr = new ArrayList<>();

            for (int i = 0; i < n; i++) {
                arr.add(sc.nextInt());
            }

            List<Integer> result = maxSubarray(arr);

            System.out.println(result.get(0) + " " + result.get(1));
        }

        sc.close();
    }
}
